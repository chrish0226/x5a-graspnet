cmake_minimum_required(VERSION 3.8)
project(x5a_parallel_ik)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(std_srvs REQUIRED)
find_package(x5a_task_interfaces REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(parallel_ik_server src/parallel_ik_server.cpp)
target_compile_features(parallel_ik_server PUBLIC cxx_std_17)
target_compile_options(parallel_ik_server PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(parallel_ik_server rclcpp moveit_core moveit_ros_planning
  moveit_msgs tf2_eigen std_srvs x5a_task_interfaces)
target_link_libraries(parallel_ik_server OpenSSL::Crypto)
install(TARGETS parallel_ik_server DESTINATION lib/${PROJECT_NAME})
ament_package()
